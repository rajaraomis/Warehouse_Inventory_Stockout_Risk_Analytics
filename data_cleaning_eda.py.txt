import pandas as pd
import numpy as np

def clean_and_analyze_inventory(file_path):
    print("--- 1. Loading Dataset ---")
    df = pd.read_csv(file_path)
    print(f"Dataset Shape: {df.shape[0]} rows, {df.shape[1]} columns\n")
    
    print("--- 2. Data Cleaning & Inspection ---")
    print("Missing values per column:")
    print(df.isnull().sum())
    
    # Fill or drop missing values
    df.ffill(inplace=True)
    
    # Ensure correct data types
    if 'date' in df.columns:
        df['date'] = pd.to_datetime(df['date'])
        
    print("\n--- 3. Exploratory Data Analysis (EDA) ---")
    total_products = df['product_id'].nunique()
    print(f"Total Unique Products: {total_products}")
    
    if 'stockout_risk' in df.columns:
        high_risk_count = df[df['stockout_risk'].isin(['Yes', 'True', 1, True])].shape[0]
        print(f"High Risk Products Count: {high_risk_count}")
        
    if 'lead_time_days' in df.columns:
        avg_lead_time = df['lead_time_days'].mean()
        print(f"Average Lead Time (Days): {round(avg_lead_time, 2)}")
        
    print("\n--- 4. Saving Cleaned Data ---")
    df.to_csv("cleaned_data.csv", index=False)
    print("Cleaned data saved as 'cleaned_data.csv'")

if __name__ == "__main__":
    clean_and_analyze_inventory("inventory_data.csv")
